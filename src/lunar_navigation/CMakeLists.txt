cmake_minimum_required(VERSION 3.8)
project(lunar_navigation)
find_package(ament_cmake REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters segmentation sample_consensus)
get_filename_component(LUNAR_WORKSPACE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(pcd_to_grid src/pcd_to_grid.cpp)
target_compile_features(pcd_to_grid PRIVATE cxx_std_17)
target_include_directories(pcd_to_grid SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(pcd_to_grid ${PCL_LIBRARIES})
target_compile_definitions(pcd_to_grid PRIVATE LUNAR_WORKSPACE_ROOT="${LUNAR_WORKSPACE_ROOT}")
install(TARGETS pcd_to_grid DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
